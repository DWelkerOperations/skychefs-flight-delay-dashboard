WITH source_rows AS (
  SELECT upper(substr(trim(station), 1, 3)) AS location,
         regexp_replace(upper(trim(oflt)), '\s+', '', 'g') AS inbound_flight,
         regexp_replace(upper(trim(flight)), '\s+', '', 'g') AS outbound_flight,
         try_strptime('2026/' || trim(sta), '%Y/%m/%d %H:%M') AS scheduled_arrival,
         try_strptime('2026/' || trim(bta), '%Y/%m/%d %H:%M') AS actual_arrival,
         try_strptime('2026/' || trim(std), '%Y/%m/%d %H:%M') AS scheduled_departure,
         try_strptime('2026/' || trim(btd), '%Y/%m/%d %H:%M') AS actual_departure,
         nullif(trim(rou), '') AS inbound_route,
         nullif(trim(rou_outbound), '') AS outbound_route
  FROM read_csv('All Tasks - Historical Data Testing 2 (3).csv', header=true, all_varchar=true)
), eligible_movements AS (
  SELECT location, inbound_flight AS flight,
         regexp_extract(inbound_flight, '^([A-Z0-9]{2})[0-9]+', 1) AS airline,
         scheduled_arrival AS scheduled_time, actual_arrival AS actual_time, 'Arrival' AS direction
  FROM source_rows
  WHERE inbound_route IS NOT NULL
  UNION ALL
  SELECT location, outbound_flight,
         regexp_extract(outbound_flight, '^([A-Z0-9]{2})[0-9]+', 1),
         scheduled_departure, actual_departure, 'Departure'
  FROM source_rows
  WHERE outbound_route IS NOT NULL
)
SELECT location, airline, direction, date_trunc('month', scheduled_time) AS month,
       count(*) AS scheduled_movements,
       count(*) FILTER (WHERE date_diff('minute', scheduled_time, actual_time) BETWEEN -180 AND 1440) AS valid_observations,
       count(*) FILTER (WHERE date_diff('minute', scheduled_time, actual_time) BETWEEN 30 AND 1440) AS delay_30_count,
       count(*) FILTER (WHERE date_diff('minute', scheduled_time, actual_time) BETWEEN 60 AND 1440) AS delay_60_count,
       count(*) FILTER (WHERE date_diff('minute', scheduled_time, actual_time) BETWEEN 90 AND 1440) AS delay_90_count
FROM eligible_movements
WHERE scheduled_time::date BETWEEN DATE '2026-02-01' AND DATE '2026-07-20'
GROUP BY location, airline, direction, month

Definitions and caveats

Arrival delay is BTA minus STA; departure delay is BTD minus STD. Delay thresholds are inclusive. Rates use valid, deduplicated movements as the denominator. Missing actual times and timestamp differences outside −180 to +1,440 minutes are excluded. Times are assumed to be local as recorded. Airline filters use the two-character flight-number prefix, so marketed flights may include codeshares. Location uses the first three station-label characters to merge aliases such as ORD and ORD NEW (MC5).
